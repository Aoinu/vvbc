    var flight = root.AddComponent<BallBehaviour>();
    var contact = root.AddComponent<BatContactBehaviour>();
    var motion = root.AddComponent<BatMotionBehaviour>();
    feedback.flight = flight; feedback.contact = contact; contact.motion = motion; contact.flight = flight;
    contact.contactZone = root.transform; contact.zoneHalfSize = new Vector3(1f, 100f, 1f);
    contact.centerDirection = Vector3.forward; contact.maxExtrapolation = 0.025f;
    flight.flying = true; flight.flightSequence = 123;
    var phantomObjects = new GameObject[7]; feedback.batPhantoms = new Transform[7]; feedback.batRenderers = new Renderer[7];
    var material = AssetDatabase.LoadAssetAtPath<Material>("Packages/org.vvbc.core/Runtime/Materials/BallPhantom.mat");

    }

    System.Func<object,string,object> get = (o,n) => o.GetType().GetField(n, flags).GetValue(o);
    System.Action<object,string> call = (o,n) => o.GetType().GetMethod(n,flags).Invoke(o,null);

    Vector3[] bp = new Vector3[64], bv = new Vector3[64], spin = new Vector3[64];
    Vector3[] mp = new Vector3[32], mv = new Vector3[32], mo = new Vector3[32], omega = new Vector3[32], referenceVelocity = new Vector3[32];

    set(motion,"_contactCoreOffsets",mo); set(motion,"_contactAngularVelocities",omega); set(motion,"_contactRotations",mr);

    for (int n = 0; n <= 90; n++)

        bt[bi] = time; bp[bi] = new Vector3(0f, 0f, -2f + 0.04f * n); bv[bi] = Vector3.forward * 2f;
        mt[mi] = time; mp[mi] = new Vector3(n * 0.01f, 1f, 0f); mv[mi] = Vector3.right * 0.5f;
        mo[mi] = new Vector3(0f, 0.3f, 0f); mr[mi] = Quaternion.Euler(0f, n * 0.5f, -90f);
        referenceVelocity[mi] = Vector3.right * 10f;
        int bc = Mathf.Min(n + 1, 64), mc = Mathf.Min(n + 1, 32);
        set(flight,"_sampleCount",bc); set(flight,"_nextSample",(bi + 1) % 64);
        flight.oldestMotionTime = bt[(bi + 1 - bc + 64) % 64]; flight.latestMotionTime = time;
        set(motion,"_contactCount",mc); set(motion,"_contactNext",(mi + 1) % 32);
        motion.contactOldestTime = mt[(mi + 1 - mc + 32) % 32]; motion.contactLatestTime = time;
        feedback.OnFeedbackBatPose();
        feedback.Update();
    }
    check((bool)get(feedback,"_hasResult") && !(bool)get(feedback,"_isContact"), "miss result");
    check(!contact.feedbackTrailActive, "capture stops after miss snapshots");
    check((bool)get(feedback,"_trailEntered") && (bool)get(feedback,"_trailEndKnown"), "entry and exit");
    float entry = (float)get(feedback,"_trailEntry"), exit = (float)get(feedback,"_trailExit");
    check(Mathf.Abs(entry - (start + 25 * dt)) < 0.0002f && Mathf.Abs(exit - (start + 75 * dt)) < 0.0002f, "actual crossing times");
    check((int)get(feedback,"_poseCount") > 32, "retains older samples");
    set(feedback,"_time",start + 50 * dt); set(feedback,"_trailReady",false); call(feedback,"FreezeTrail");
    var valid = (bool[])get(feedback,"_trailValid"); var positions = (Vector3[])get(feedback,"_trailPositions");
    var rotations = (Quaternion[])get(feedback,"_trailRotations"); var colors = (Color[])get(feedback,"_trailColors");
    for (int i = 0; i < 7; i++)
    {
        float n = 25f + 50f * i / 6f;
        Quaternion expectedRotation = Quaternion.Euler(0f, n * 0.5f, -90f);
        Vector3 expected = new Vector3(n * 0.01f, 1f, 0f) - expectedRotation * new Vector3(0f, 0.3f, 0f);
        check(valid[i] && Vector3.Distance(positions[i], expected) < 0.002f && Quaternion.Angle(rotations[i], expectedRotation) < 0.1f, "pose " + i);
        check(i == 0 || colors[i] != colors[0], "independent timing color " + i);
    }
    call(feedback, "ShowBatTrail");
    check(feedback.trailValid == 7 && feedback.trailRendered == 7, "seven render calls");
    for (int i = 0; i < 7; i++)
        check(feedback.batRenderers[i].enabled && Vector3.Distance(feedback.batPhantoms[i].position, positions[i]) < 0.0001f &&
            Quaternion.Angle(feedback.batPhantoms[i].rotation, rotations[i]) < 0.01f &&
            feedback.batRenderers[i].material.color == colors[i], "rendered pose and color " + i);
    for (int i = 0; i < 7; i++) feedback.batRenderers[i].enabled = false;
    var goodTransforms = feedback.batPhantoms; var goodRenderers = feedback.batRenderers;
    feedback.batPhantoms = new Transform[0]; feedback.batRenderers = new Renderer[0];
    set(feedback, "_trailReady", false); call(feedback, "FreezeTrail");
    check(feedback.trailAvailable == 0 && feedback.trailRequested == 7 && feedback.trailRejectionReason != "" && (int)get(feedback,"_trailCount") == 0, "empty arrays rejected");
    feedback.batPhantoms = goodTransforms; feedback.batRenderers = (Renderer[])goodRenderers.Clone(); feedback.batRenderers[3] = null;
    set(feedback, "_trailReady", false); call(feedback, "FreezeTrail");
    check(feedback.trailAvailable == 6 && feedback.trailRejectionReason != "", "missing element rejected");
    feedback.batRenderers = goodRenderers;
    // A discontinuity invalidates only times inside its interval.
    var segments = (int[])get(feedback,"_poseSegments");
    var poseTimes = (float[])get(feedback,"_poseTimes");
    int poseCount = (int)get(feedback,"_poseCount");
    for (int i = 0; i < poseCount; i++) segments[i] = poseTimes[i] >= start + 0.68f ? 1 : 0;
    set(feedback, "_trailReady", false); call(feedback, "FreezeTrail"); call(feedback, "ShowBatTrail");
    check(feedback.trailValid > 0 && feedback.trailValid < 7 && feedback.trailRendered == feedback.trailValid, "gap preserves other rendered poses");
    for (int i = 0; i < poseCount; i++) segments[i] = 0;
    for (int i = 0; i < 7; i++) feedback.batRenderers[i].enabled = false;
    feedback.phantomCount = 1; set(feedback, "_trailReady", false); call(feedback, "FreezeTrail");
    var single = (Vector3[])get(feedback, "_trailPositions");
    var singleRotation = Quaternion.Euler(0f, 25f, -90f);
    check((int)get(feedback, "_trailCount") == 1 && ((bool[])get(feedback, "_trailValid"))[0] &&
        Vector3.Distance(single[0], new Vector3(0.5f, 1f, 0f) - singleRotation * new Vector3(0f, 0.3f, 0f)) < 0.002f, "single reference time");
    feedback.phantomCount = 7;
    // Exercise the public contact notification using the already-captured
    // incoming trajectory, after more than one Motion ring's worth of samples.
    set(feedback, "_trailReady", false);
    contact.contactTime = start + 50f * dt; contact.contactPosition = Vector3.zero;
    contact.snapshotIncomingVelocity = Vector3.forward * 2f; contact.snapshotIncomingSpin = Vector3.zero;
    contact.resultVelocity = Vector3.right * 100f; contact.resultSpin = Vector3.up * 30f;
    contact.contactValid = true; contact.contactPitchSequence = flight.flightSequence;
    flight.dragCoefficient = 0f; flight.magnusCoefficient = 0f; flight.contactConsumed = true;
    feedback.OnContactConfirmed();
    check(contact.feedbackTrailActive, "contact resumes capture");
    check((bool)get(feedback, "_hasResult") && (bool)get(feedback, "_isContact") &&
        (bool)get(feedback, "_trailEndKnown") && Mathf.Abs((float)get(feedback, "_trailExit") - exit) < 0.002f, "contact predicts pitch exit");
    call(feedback, "FreezeTrail");
    check(((bool[])get(feedback,"_trailValid")).All(v => v), "seven contact poses");
    call(feedback, "ShowBatTrail");
    check(feedback.trailRendered == 7 && feedback.batRenderers.All(r => r.enabled), "contact renders seven");
    check((float)get(feedback, "_showAt") > Time.time + 0.9f, "one second delay");
    feedback.Update(); check(!(bool)get(feedback, "_shown"), "no early display");
    for (int i = 0; i < 7; i++) feedback.batRenderers[i].enabled = true;
    feedback.OnBallReset();
    check(!contact.feedbackTrailActive && feedback.batRenderers.All(r => !r.enabled), "reset clears renderers and capture");
    set(feedback,"_trailEntered",true); set(feedback,"_trailEntry",start);
    contact.contactTime = start; contact.contactPosition = Vector3.zero;
    contact.snapshotIncomingVelocity = Vector3.forward * 2f; contact.snapshotIncomingSpin = Vector3.zero;
    flight.dragCoefficient = 0f; flight.magnusCoefficient = 0f;
    contact.resultVelocity = Vector3.right * 100f; call(feedback,"PredictTrailExit");
    float predicted = (float)get(feedback,"_trailExit");
    check((bool)get(feedback,"_trailEndKnown") && Mathf.Abs(predicted - start - 0.5f) < 0.002f, "predicted exit");
    contact.resultVelocity = Vector3.back * 500f; contact.resultSpin = Vector3.up * 300f;
    call(feedback,"PredictTrailExit");

    // Default 3 m deep zone at 140 km/h, at common VR display rates.
    foreach (float frameRate in new[] { 72f, 90f, 120f })

        flight.flightSequence++; flight.contactConsumed = false; flight.flying = true;
        contact.zoneHalfSize = new Vector3(1.5f, 100f, 1.5f);
        feedback.OnFlightStarted();

        for (int n = 0; n < 20; n++)
        {
            float t = fastStart + n * frame; int bi = n % 64, mi = n % 32;
            bt[bi] = t; bp[bi] = new Vector3(0f, 0f, -2f + velocity * n * frame); bv[bi] = Vector3.forward * velocity;
            mt[mi] = t; mp[mi] = new Vector3(n * frame, 1f, 0f); mv[mi] = Vector3.right;
            mo[mi] = Vector3.zero; mr[mi] = Quaternion.Euler(0f, n * 2f, -90f); referenceVelocity[mi] = Vector3.right * 10f;
            int bc = Mathf.Min(n + 1, 64), mc = Mathf.Min(n + 1, 32);
            set(flight,"_sampleCount",bc); set(flight,"_nextSample",(bi + 1) % 64);
            flight.oldestMotionTime = bt[(bi + 1 - bc + 64) % 64]; flight.latestMotionTime = t;
            set(motion,"_contactCount",mc); set(motion,"_contactNext",(mi + 1) % 32);
            motion.contactOldestTime = mt[(mi + 1 - mc + 32) % 32]; motion.contactLatestTime = t;
            feedback.OnFeedbackBatPose();
            if (hit && !flight.contactConsumed && t >= fastStart + 2f / velocity)
            {
                contact.feedbackCandidateTime = fastStart + 2f / velocity;
                feedback.OnFeedbackPreContact();
                contact.contactTime = contact.feedbackCandidateTime; contact.contactPosition = Vector3.zero;
                contact.snapshotIncomingVelocity = Vector3.forward * velocity; contact.snapshotIncomingSpin = Vector3.zero;
                contact.contactValid = true; contact.contactPitchSequence = flight.flightSequence;
                flight.contactConsumed = true; feedback.OnContactConfirmed();
            }
            feedback.Update();
        }
        check((bool)get(feedback,"_hasResult") && (bool)get(feedback,"_isContact") == hit, "fast result " + frameRate + " hit=" + hit);
        check(feedback.trailValid == 7, "fast measured samples " + frameRate);
        call(feedback,"ShowBatTrail");
        check(feedback.trailRendered == 7, "fast render " + frameRate);
        feedback.OnBallReset();
    }
    return "PASS seven rendered poses/colors, invalid references, partial gap, contact/miss, reset, 140 km/h contact/miss at 72/90/120 Hz";

                if (usedMaterial != null && !EditorUtility.IsPersistent(usedMaterial)) temporaryMaterials.Add(usedMaterial);

    foreach (var usedMaterial in temporaryMaterials) UnityEngine.Object.DestroyImmediate(usedMaterial);
